"""Bind only the REAL imported original-card Grand skeletal assets to Luminous.

Requires all 57 original Unity renderer meshes to exist in the ACTUAL game
source tree. Changes the single original character-level motion binding; does

import hashlib
import json
from pathlib import Path

ROOT=Path(r"D:\UnrealProjects\Gate12_WholeCharacter_Build_20261031")
SRC=Path(r"D:\UnrealProjects\G12_LuminousNativeImport_20261031\AuthenticClothedOriginalCard3D\Grand")
SOURCE=SRC/"GRAND_EXACT_ORIGINAL_SOURCE_BLENDER_MANIFEST.json"
GAME=ROOT/"Content/NoWorld/Characters/Luminous/Grand/ExactOriginalClothed"
CPP=ROOT/"Source/OfflineGame/Private/Koikatsu/OGExactOriginalLuminousParts.cpp"
CARD_HASH="26c48e1c03c154f9f39a0b06afaade53423c11dbd079604e430162eb4cae5353"

source=json.loads(SOURCE.read_text("utf8"))
assert source["card_sha256"]==CARD_HASH and source["original_active_renderer_count"]==57
groups=source["groups"]
assert len(groups)==9
all_source_indices=[]
native=[]
for group in groups:
    i=group["id"]
    ids=sorted(group["original_renderer_indices"])
    assert 0<=i<9 and ids
    assert sum(int(s["vertices"]) for s in source["source_renderers"] if s["renderer"] in ids)==group["source_vertices"]
    all_source_indices.extend(ids)
    folder=GAME/("Part_%02d"%i)
    assert folder.is_dir(), "The actual game is missing Grand original group %02d"%i
    skeletal=sorted(p for p in folder.rglob("*.uasset") if
        "SkeletalMeshes" in p.parts and
        not p.stem.endswith(("_Skeleton","_PhysicsAsset")))
    skeleton=sorted(p for p in folder.rglob("*_Skeleton.uasset"))
    assert len(skeletal)==len(ids), "Unreal did not import every renderer in Grand group %02d: %d of %d"%(i,len(skeletal),len(ids))
    assert skeleton,"Original source group %02d has no native skeletal rig"%i
    if len(ids)>1:
        for idx in ids:
            name="Grand_ExactRenderer_%03d"%idx
            candidates=[p for p in skeletal if p.stem==name]
            assert len(candidates)==1, "Missing exact original renderer %03d native asset"%idx
            p=candidates[0]
            native.append((idx,p))
    else:
        native.append((ids[0],skeletal[0]))
assert len(all_source_indices)==57 and len(set(all_source_indices))==57
assert set(all_source_indices)==set(s["renderer"] for s in source["source_renderers"])
assert len(native)==57 and len({str(p) for _,p in native})==57
native.sort(key=lambda x:x[0])

    uri="/Game/"+relative.as_posix()+"."+p.stem
